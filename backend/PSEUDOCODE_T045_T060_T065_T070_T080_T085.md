## T045: Meeting State Transition to PROCESSING

**File:** `backend/src/services/meetings/meeting.service.ts`

```typescript
class MeetingService {
  /**
   * Transition meeting to PROCESSING state
   * Called after meeting ends if recording/transcription enabled
   * Queues async processing jobs
   */
  async transitionToProcessing(
    meetingId: string | Types.ObjectId,
    meeting: IMeeting,
  ): Promise<void> {
    // 1. Validate current state is ENDED
    if (meeting.status !== MeetingStatus.ENDED) {
      throw ApiError.badRequest(`Meeting must be ENDED to transition to PROCESSING`);
    }

    // 2. Validate transition
    validateMeetingTransition(MeetingStatus.ENDED, MeetingStatus.PROCESSING);

    // 3. Update meeting status
    meeting.status = MeetingStatus.PROCESSING;
    await meeting.save();

    // 4. Queue transcription job if recording enabled
    if (meeting.recordingConfig.enabled) {
      const recording = await Recording.findOne({ meetingId });
      if (recording) {
        await getTranscriptionQueue().add("transcribe", {
          recordingId: recording._id,
          meetingId,
          provider: "deepgram", // From env
        }, {
          attempts: 3,
          backoff: { type: "exponential", delay: 3000 },
        });
      }
    }

    // 5. Queue AI MoM generation if enabled
    if (meeting.aiConfig.enabled) {
      await getAiMomQueue().add("generate", {
        meetingId,
        provider: "openai", // From env
      }, {
        attempts: 2,
        backoff: { type: "exponential", delay: 5000 },
      });
    }

    // 6. Log audit event
    await auditLogsService.logAction({
      organizationId: meeting.organizationId,
      actorId: meeting.hostId, // System action
      action: AuditAction.MEETING_UPDATED,
      targetType: "Meeting",
      targetId: meeting._id,
      status: "success",
      metadata: {
        previousStatus: MeetingStatus.ENDED,
        newStatus: MeetingStatus.PROCESSING,
        transcriptionQueued: meeting.recordingConfig.enabled,
        aiProcessingQueued: meeting.aiConfig.enabled,
      },
    });

    // 7. Send notification
    // Notify host: "Meeting processing has started. Minutes will be ready for review soon."
  }
}
```

```typescript
class MinutesService {
  /**
   * Approve minutes (DRAFT → REVIEW → APPROVED → PUBLISHED)
   * Each transition creates new version in immutable history
   */
  async approveMinutes(
    minutesId: string | Types.ObjectId,
    userId: string | Types.ObjectId,
    comments?: string,
  ): Promise<void> {
    const minutesDoc = await MinutesDocument.findById(minutesId);

    // Validate current status
    if (minutesDoc.currentStatus !== MinutesStatus.REVIEW) {
      throw ApiError.badRequest(`Minutes must be in REVIEW to approve`);
    }

    // Create new version with APPROVED status
    const newVersion: MinutesVersion = {
      version: minutesDoc.versions.length,
      createdBy: userId,
      createdAt: new Date(),
      content: minutesDoc.versions[minutesDoc.currentVersion].content,
      status: MinutesStatus.APPROVED,
      approvedBy: userId,
      revisionNotes: comments,
    };

    minutesDoc.versions.push(newVersion);
    minutesDoc.currentStatus = MinutesStatus.APPROVED;
    minutesDoc.currentVersion = newVersion.version;
    minutesDoc.lastModifiedBy = userId;
    minutesDoc.lastModifiedAt = new Date();
    await minutesDoc.save();

    // Log audit event
    await auditLogsService.logAction({
      organizationId: minutesDoc.organizationId,
      actorId: userId,
      action: AuditAction.MINUTES_APPROVED,
      targetType: "MinutesDocument",
      targetId: minutesDoc._id,
      status: "success",
      metadata: { version: newVersion.version, comments },
    });
  }

  /**
   * Publish minutes (APPROVED → PUBLISHED)
   * Official record, visible to all participants
   */
  async publishMinutes(
    minutesId: string | Types.ObjectId,
    userId: string | Types.ObjectId,
  ): Promise<void> {
    const minutesDoc = await MinutesDocument.findById(minutesId);

    if (minutesDoc.currentStatus !== MinutesStatus.APPROVED) {
      throw ApiError.badRequest("Minutes must be APPROVED to publish");
    }

    // Create published version
    const currentVersion = minutesDoc.versions[minutesDoc.currentVersion];
    const publishedVersion: MinutesVersion = {
      ...currentVersion,
      version: minutesDoc.versions.length,
      status: MinutesStatus.PUBLISHED,
      publishedAt: new Date(),
    };

    minutesDoc.versions.push(publishedVersion);
    minutesDoc.currentStatus = MinutesStatus.PUBLISHED;
    minutesDoc.publishedVersion = publishedVersion.version;
    minutesDoc.currentVersion = publishedVersion.version;
    minutesDoc.lastModifiedBy = userId;
    minutesDoc.lastModifiedAt = new Date();
    await minutesDoc.save();

    // Update meeting status to COMPLETED
    await Meeting.updateOne(
      { _id: minutesDoc.meetingId },
      { status: MeetingStatus.COMPLETED },
    );

    // Log audit event
    await auditLogsService.logAction({
      organizationId: minutesDoc.organizationId,
      actorId: userId,
      action: AuditAction.MINUTES_PUBLISHED,
      targetType: "MinutesDocument",
      targetId: minutesDoc._id,
      status: "success",
      metadata: { publishedVersion: publishedVersion.version },
    });

    // Notify all participants
    // Send published minutes to meeting participants
  }

  /**
   * Revise minutes (PUBLISHED → new DRAFT version)
   * Previous versions preserved for audit trail
   */

    minutesId: string | Types.ObjectId,
    newContent: string,
    userId: string | Types.ObjectId,
    revisionReason: string,
  ): Promise<void> {
    const minutesDoc = await MinutesDocument.findById(minutesId);

    const revisionVersion: MinutesVersion = {
      version: minutesDoc.versions.length,
      createdBy: userId,

    minutesDoc.currentStatus = MinutesStatus.DRAFT;

    await auditLogsService.logAction({

      action: AuditAction.MINUTES_REVISED,
      targetType: "MinutesDocument",
      targetId: minutesDoc._id,
      status: "success",
      metadata: {
        previousVersion: minutesDoc.publishedVersion,
        revisionVersion: revisionVersion.version,
        reason: revisionReason,
```

```typescript
class ActionItemService {
  /**
   * Update action item with audit trail
   */
  async updateActionItem(
    itemId: string | Types.ObjectId,
    updates: { ownerId?: Types.ObjectId; dueAt?: Date; status?: string },
    userId: string | Types.ObjectId,
    reason?: string,
  ): Promise<void> {
    const item = await ActionItem.findById(itemId);

    // Track each change
    if (updates.ownerId && updates.ownerId.toString() !== item.ownerId.toString()) {
      item.auditTrail.push({
        field: "ownership",
        oldValue: item.ownerId,
        newValue: updates.ownerId,
        changedBy: userId,
        changedAt: new Date(),
        reason,
      });
      item.ownerId = updates.ownerId;
    }

    if (updates.dueAt && updates.dueAt !== item.dueAt) {
      item.auditTrail.push({
        field: "dueDate",
        oldValue: item.dueAt,
        newValue: updates.dueAt,
        changedBy: userId,
        changedAt: new Date(),
        reason,
      });
      item.dueAt = updates.dueAt;
    }

    if (updates.status && updates.status !== item.status) {
      item.auditTrail.push({
        field: "status",
        oldValue: item.status,
        newValue: updates.status,
        changedBy: userId,
        changedAt: new Date(),
        reason,
      });
      item.status = updates.status;

      // Log audit event
      await auditLogsService.logAction({
        organizationId: item.organizationId,
        actorId: userId,
        action: AuditAction.ACTION_ITEM_STATUS_CHANGED,
        targetType: "ActionItem",
        targetId: item._id,
        status: "success",
        metadata: { oldStatus: item.status, newStatus: updates.status },
      });
    }

    await item.save();
  }
}
```
